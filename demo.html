class SmartNotificationListener : NotificationListenerService() {
    override fun onNotificationPosted(sbn: StatusBarNotification) {
        val extras = sbn.notification.extras
        val pkg = sbn.packageName

        // 1. Circular Download Progress Ring around Pixel 10a Notch
        val max = extras.getInt(Notification.EXTRA_PROGRESS_MAX, 0)
        val progress = extras.getInt(Notification.EXTRA_PROGRESS, 0)
        val indeterminate = extras.getBoolean(Notification.EXTRA_PROGRESS_INDETERMINATE, false)
        if (max > 0 && !indeterminate) {
            IslandState.downloadProgress = (progress.toFloat() / max.toFloat()).coerceIn(0f, 1f)
            IslandState.mode = IslandMode.DOWNLOAD
            return
        }

        // 2. Google Maps Live Turn-by-Turn Navigation
        if (pkg == "com.google.android.apps.maps") {
            val title = extras.getString(Notification.EXTRA_TITLE) ?: "Next turn ahead"
            IslandState.navInstruction = title
            IslandState.mode = IslandMode.MAPS_NAV
            return
        }

        // 3. Instagram DM & WhatsApp Mirroring (Like + Inline Reply)
        if (pkg == "com.instagram.android" || pkg == "com.whatsapp") {
            IslandState.sender = extras.getString(Notification.EXTRA_TITLE) ?: return
            IslandState.message = extras.getCharSequence(Notification.EXTRA_TEXT)?.toString() ?: return

            sbn.notification.actions?.forEach { action ->
                val title = action.title?.toString()?.lowercase() ?: ""
                if (title.contains("like") || title.contains("❤")) {
                    IslandState.likeActionIntent = action.actionIntent
                }
                if (!action.remoteInputs.isNullOrEmpty()) {
                    IslandState.replyAction = action
                }
            }
            IslandState.mode = if (pkg == "com.instagram.android") IslandMode.INSTA_DM else IslandMode.WHATSAPP
            IslandState.isExpanded = true
        }
    }
}

fun sendInlineReply(context: Context, replyText: String) {
    val action = IslandState.replyAction ?: return
    val remoteInputs = action.remoteInputs ?: return
    val intent = Intent()
    val bundle = Bundle()
    for (input in remoteInputs) {
        bundle.putCharSequence(input.resultKey, replyText)
    }
    RemoteInput.addResultsToIntent(remoteInputs, intent, bundle)
    action.actionIntent.send(context, 0, intent)
    IslandState.isExpanded = false
}

@Composable
fun NotchCircularProgressRing(progress: Float, ringColor: Color = IslandState.accentColor) {
    val view = LocalView.current
    Box(
        contentAlignment = Alignment.Center,
        modifier = Modifier
            .size(26.dp)
            .clickable {
                view.performHapticFeedback(HapticFeedbackConstants.CONTEXT_CLICK)
                IslandState.isExpanded = !IslandState.isExpanded
            }
    ) {
        Canvas(modifier = Modifier.fillMaxSize()) {
            drawArc(
                color = ringColor.copy(alpha = 0.2f),
                startAngle = -90f,
                sweepAngle = 360f,
                useCenter = false,
                style = Stroke(width = 2.5.dp.toPx(), cap = StrokeCap.Round)
            )
            drawArc(
                color = ringColor,
                startAngle = -90f,
                sweepAngle = 360f * progress,
                useCenter = false,
                style = Stroke(width = 2.5.dp.toPx(), cap = StrokeCap.Round)
            )
        }
    }
}

          object IslandController {
              var overlayView: FrameLayout? = null
              var windowManager: WindowManager? = null
              var layoutParams: WindowManager.LayoutParams? = null
              var isExpanded = false
              var compactWidthDp = 132
              var topOffsetDp = 5
              var downloadPct = -1
              var titleText = "Pixel 10a Island"
              var subText = "Tap pill to expand • Swipe to dismiss"
              var rightBadgeText = "Active"
              var likeIntent: PendingIntent? = null
              var replyAction: Notification.Action? = null

              fun dp(ctx: Context, v: Int): Int =
                  TypedValue.applyDimension(TypedValue.COMPLEX_UNIT_DIP, v.toFloat(), ctx.resources.displayMetrics).toInt()

              fun showOrUpdate(ctx: Context, autoExpand: Boolean = false) {
                  if (!Settings.canDrawOverlays(ctx)) return
                  val appCtx = ctx.applicationContext
                  val wm = (windowManager ?: appCtx.getSystemService(Context.WINDOW_SERVICE) as WindowManager).also { windowManager = it }

                  if (autoExpand) isExpanded = true

                  if (overlayView == null) {
                      val lp = WindowManager.LayoutParams(
                          dp(appCtx, compactWidthDp),
                          dp(appCtx, 30),
                          WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
                          WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
                              WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN or
                              WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
                          PixelFormat.TRANSLUCENT
                      ).apply {
                          gravity = Gravity.TOP or Gravity.CENTER_HORIZONTAL
                          y = dp(appCtx, topOffsetDp)
                      }
                      layoutParams = lp
                      val container = createIslandView(appCtx)
                      overlayView = container
                      wm.addView(container, lp)
                  }
                  animateMorph(appCtx)
              }

              private fun createIslandView(ctx: Context): FrameLayout {
                  val root = object : FrameLayout(ctx) {
                      private val ringPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
                          style = Paint.Style.STROKE
                          strokeWidth = dp(ctx, 3).toFloat()
                          strokeCap = Paint.Cap.ROUND
                          color = Color.parseColor("#38BDF8")
                      }
                      override fun dispatchDraw(canvas: Canvas) {
                          super.dispatchDraw(canvas)
                          if (downloadPct in 0..100 && !isExpanded) {
                              val cx = width / 2f
                              val cy = height / 2f
                              val r = dp(ctx, 11).toFloat()
                              val rect = RectF(cx - r, cy - r, cx + r, cy + r)
                              ringPaint.alpha = 60
                              canvas.drawArc(rect, 0f, 360f, false, ringPaint)
                              ringPaint.alpha = 255
                              canvas.drawArc(rect, -90f, 360f * (downloadPct / 100f), false, ringPaint)
                          }
                      }
                  }
                  val bg = GradientDrawable().apply {
                      setColor(Color.BLACK)
                      cornerRadius = dp(ctx, 50).toFloat()
                      setStroke(dp(ctx, 1), Color.parseColor("#3338BDF8"))
                  }
                  root.background = bg

                  var downX = 0f
                  var dragged = false
                  root.setOnTouchListener { v, ev ->
                      when (ev.actionMasked) {
                          MotionEvent.ACTION_DOWN -> { downX = ev.rawX; dragged = false; true }
                          MotionEvent.ACTION_MOVE -> {
                              if (abs(ev.rawX - downX) > dp(ctx, 18)) dragged = true
                              true
                          }
                          MotionEvent.ACTION_UP -> {
                              if (dragged && isExpanded) {
                                  isExpanded = false
                                  animateMorph(ctx)
                              } else if (!dragged) {
                                  v.performHapticFeedback(HapticFeedbackConstants.CONTEXT_CLICK)
                                  isExpanded = !isExpanded
                                  animateMorph(ctx)
                              }
                              true
                          }
                          else -> false
                      }
                  }
                  rebuildInnerContent(ctx, root)
                  return root
              }

              private fun rebuildInnerContent(ctx: Context, root: FrameLayout) {
                  root.removeAllViews()
                  val col = LinearLayout(ctx).apply {
                      orientation = LinearLayout.VERTICAL
                      setPadding(dp(ctx, 12), dp(ctx, 5), dp(ctx, 12), dp(ctx, 8))

              private fun sendQuickReply(ctx: Context, msg: String) {
                  val act = replyAction ?: return
                  val inputs = act.remoteInputs ?: return
                  val intent = Intent()
                  val bundle = Bundle()
                  for (inp in inputs) bundle.putCharSequence(inp.resultKey, msg)
                  RemoteInput.addResultsToIntent(inputs, intent, bundle)
                  try { act.actionIntent.send(ctx, 0, intent) } catch (_: Exception) {}
              }

              fun animateMorph(ctx: Context) {
                  val view = overlayView ?: return
                  val lp = layoutParams ?: return
                  val wm = windowManager ?: return
                  rebuildInnerContent(ctx, view)

                  val startW = lp.width
                  val startH = lp.height
                  val endW = dp(ctx, if (isExpanded) 300 else compactWidthDp)
                  val endH = dp(ctx, if (isExpanded) 134 else 30)

                  ValueAnimator.ofFloat(0f, 1f).apply {
                      duration = 420
                      interpolator = OvershootInterpolator(1.15f)
                      addUpdateListener { anim ->
                          val f = anim.animatedValue as Float
                          lp.width = (startW + (endW - startW) * f).toInt()
                          lp.height = (startH + (endH - startH) * f).toInt()
                          lp.y = dp(ctx, topOffsetDp)
                          try { wm.updateViewLayout(view, lp) } catch (_: Exception) {}
                      }
                      start()
                  }
              }
          }
